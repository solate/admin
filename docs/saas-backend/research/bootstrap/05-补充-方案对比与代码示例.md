# 05-补充：方案对比与代码示例

## 方案 A vs 方案 B 代码对比

### 方案 A：Scheduler 作为 Server 字段（推荐）

```go
// internal/server/server.go
type Server struct {
    httpSrv         *http.Server
    scheduler       *xcron.Scheduler  // ← 调度器字段
    db              *gorm.DB
    rdb             *redis.Client
    gracefulTimeout time.Duration
}

type Options struct {
    Config    *config.Config
    DB        *gorm.DB
    RDB       *redis.Client
    Scheduler *xcron.Scheduler
}

func New(opts Options) (*Server, error) {
    gin.SetMode(opts.Config.Server.Mode)
    engine := gin.New()
    router.Setup(engine, opts.Config)

    httpSrv := &http.Server{
        Addr:    fmt.Sprintf(":%d", opts.Config.Server.Port),
        Handler: engine,
        // ...
    }
    
    return &Server{
        httpSrv:         httpSrv,
        scheduler:       opts.Scheduler,  // ← 注入
        db:              opts.DB,
        rdb:             opts.RDB,
        gracefulTimeout: time.Duration(opts.Config.Server.GracefulTimeout) * time.Second,
    }, nil
}

func (s *Server) Run(ctx context.Context) error {
    g, ctx := errgroup.WithContext(ctx)

    // 组件 1：HTTP server
    g.Go(func() error {
        slog.Info("http server starting", "addr", s.httpSrv.Addr)
        if err := s.httpSrv.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
            return err
        }
        return nil
    })

    g.Go(func() error {
        <-ctx.Done()
        shutdownCtx, cancel := context.WithTimeout(context.Background(), s.gracefulTimeout)
        defer cancel()
        slog.Info("http server shutting down")
        return s.httpSrv.Shutdown(shutdownCtx)
    })

    // 组件 2：Cron scheduler
    g.Go(func() error {
        return s.scheduler.Start(ctx)  // ← 阻塞，ctx 取消时自动优雅停止
    })

    return g.Wait()
}
```

```go
// cmd/server/jobs/register.go
package jobs

import (
    "admin/internal/service/operationlog"
    "admin/pkg/rbac"
    "admin/pkg/xcron"
)

type Deps struct {
    OperationLogSvc *operationlog.Service
    RBACCache       *rbac.PermissionCache
}

func Register(s *xcron.Scheduler, deps Deps) {
    // 每日凌晨 2 点清理 30 天前的操作日志
    s.AddFunc("0 0 2 * * *", func() {
        deps.OperationLogSvc.CleanOldLogs(context.Background(), 30)
    })

    // 每 5 分钟刷新一次 RBAC 权限缓存
    s.AddFunc("0 */5 * * * *", func() {
        deps.RBACCache.Reload(context.Background())
    })
}
```

```go
// cmd/server/main.go
func run() error {
    // 第一层：基础设施
    cfg, err := config.InitConfig()
    if err != nil {
        return fmt.Errorf("load config: %w", err)
    }

    log := xslog.New(xslog.Config{...})
    slog.SetDefault(log)

    db, err := xgorm.New(xgorm.Config(cfg.Database), log)
    if err != nil {
        return fmt.Errorf("connect database: %w", err)
    }
    defer xgorm.Close(db)

    rdbClient, err := xredis.New(xredis.Config(cfg.Redis))
    if err != nil {
        return fmt.Errorf("connect redis: %w", err)
    }
    defer rdbClient.Close()

    log.Info("all infrastructure initialized", slog.Int("port", cfg.Server.Port))

    // 第二层：长驻组件
    // 2a. 创建调度器（在 Server 之前，因为 jobs.Register 需要它）
    scheduler := xcron.New()

    // 2b. 注册静态任务
    jobs.Register(scheduler, jobs.Deps{
        OperationLogSvc: operationlog.NewService(db),
        RBACCache:       rbac.NewPermissionCache(db),
    })

    // 2c. 组装 Server（注入 scheduler）
    srv, err := server.New(server.Options{
        Config:    cfg,
        DB:        db,
        RDB:       rdbClient,
        Scheduler: scheduler,  // ← 注入
    })
    if err != nil {
        return fmt.Errorf("init server: %w", err)
    }

    ctx, stop := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
    defer stop()

    // 内部 errgroup 同时跑 HTTP + Scheduler
    if err := srv.Run(ctx); err != nil {
        return fmt.Errorf("run server: %w", err)
    }
    log.Info("server exited")
    return nil
}
```

**优点**：
- 生命周期统一在 Server.Run，main 不需要协调多个组件
- scheduler 和 HTTP 共享 db/rdb，访问自然（同一 struct 字段）
- 后续 HTTP API 要操作 scheduler，直接访问 `s.scheduler`（handler 通过 service 拿到引用）

**缺点**：
- Server 概念膨胀（不只是 HTTP，还管 scheduler）
- 如果要独立运行 scheduler 进程，需要重构

### 方案 B：Scheduler 和 Server 平级（Kubernetes 风格）

```go
// internal/server/server.go 保持原样，不加 scheduler 字段
type Server struct {
    httpSrv         *http.Server
    db              *gorm.DB
    rdb             *redis.Client
    gracefulTimeout time.Duration
}

func (s *Server) Run(ctx context.Context) error {
    g, ctx := errgroup.WithContext(ctx)
    // 只管 HTTP
    g.Go(func() error { ... })
    g.Go(func() error { /* 优雅关闭 */ })
    return g.Wait()
}
```

```go
// cmd/server/main.go
func run() error {
    // 第一层：基础设施
    cfg := ...
    log := ...
    db  := ...
    rdb := ...

    // 第二层：长驻组件
    srv, err := server.New(server.Options{
        Config: cfg,
        DB:     db,
        RDB:    rdb,
    })

    scheduler := xcron.New()
    jobs.Register(scheduler, jobs.Deps{...})

    ctx, stop := signal.NotifyContext(...)
    defer stop()

    // main 负责编排多组件
    g, ctx := errgroup.WithContext(ctx)
    g.Go(func() error { return srv.Run(ctx) })
    g.Go(func() error { return scheduler.Start(ctx) })
    
    if err := g.Wait(); err != nil {
        return fmt.Errorf("run components: %w", err)
    }
    log.Info("all components exited")
    return nil
}
```

**优点**：
- Server 职责单一（只管 HTTP）
- 独立进程部署时只需改 main（可以写个 cmd/scheduler/main.go，只跑 scheduler）
- 更符合 Kubernetes controller-manager 的风格

**缺点**：
- HTTP API 要操作 scheduler，需要通过 main 传引用（或 service 层全局变量）
- main 代码略复杂（多一个 errgroup）

## 动态任务管理示例（后续实现）

### Service 层维护名字映射

```go
// internal/service/cronjob/cronjob.go
package cronjob

import (
    "context"
    "sync"

    "github.com/robfig/cron/v3"
    "admin/pkg/xcron"
    "admin/pkg/xerr"
)

type Service struct {
    scheduler *xcron.Scheduler
    entries   map[string]cron.EntryID  // 名字 → EntryID 映射
    mu        sync.RWMutex
}

func NewService(scheduler *xcron.Scheduler) *Service {
    return &Service{
        scheduler: scheduler,
        entries:   make(map[string]cron.EntryID),
    }
}

// AddJob 动态添加任务
func (s *Service) AddJob(ctx context.Context, name, spec string, fn func()) error {
    s.mu.Lock()
    defer s.mu.Unlock()

    if _, exists := s.entries[name]; exists {
        return xerr.New(xerr.ErrConflict.Code, "任务已存在: "+name)
    }

    id, err := s.scheduler.AddFunc(spec, fn)
    if err != nil {
        return xerr.Wrap(xerr.ErrInvalidInput.Code, "添加任务失败", err)
    }

    s.entries[name] = id
    return nil
}

// RemoveJob 动态移除任务
func (s *Service) RemoveJob(ctx context.Context, name string) error {
    s.mu.Lock()
    defer s.mu.Unlock()

    id, exists := s.entries[name]
    if !exists {
        return xerr.New(xerr.ErrNotFound.Code, "任务不存在: "+name)
    }

    s.scheduler.Remove(id)
    delete(s.entries, name)
    return nil
}

// ListJobs 列出所有任务
func (s *Service) ListJobs(ctx context.Context) []JobInfo {
    s.mu.RLock()
    defer s.mu.RUnlock()

    entries := s.scheduler.Entries()
    jobs := make([]JobInfo, 0, len(s.entries))

    for name, id := range s.entries {
        for _, e := range entries {
            if e.ID == id {
                jobs = append(jobs, JobInfo{
                    Name: name,
                    Spec: "N/A",  // robfig/cron 不暴露 spec，需要额外存
                    Next: e.Next,
                    Prev: e.Prev,
                })
                break
            }
        }
    }
    return jobs
}

type JobInfo struct {
    Name string
    Spec string
    Next time.Time
    Prev time.Time
}
```

### Handler 层调用

```go
// internal/handler/cronjob/cronjob.go
package cronjob

import (
    "github.com/gin-gonic/gin"
    "admin/internal/service/cronjob"
    "admin/pkg/response"
)

type Handler struct {
    svc *cronjob.Service
}

func NewHandler(svc *cronjob.Service) *Handler {
    return &Handler{svc: svc}
}

// @Summary 添加定时任务
// @Tags Cron管理
// @Accept json
// @Produce json
// @Param body body AddJobRequest true "任务信息"
// @Success 200 {object} response.Response
// @Router /api/cron/jobs [post]
func (h *Handler) AddJob(c *gin.Context) {
    var req AddJobRequest
    if err := c.ShouldBindJSON(&req); err != nil {
        response.FailWithMsg(c, "参数错误: "+err.Error())
        return
    }

    // 实际项目中 fn 需要从任务注册表查找
    err := h.svc.AddJob(c.Request.Context(), req.Name, req.Spec, func() {
        // 具体任务逻辑
    })
    if err != nil {
        response.FailWithError(c, err)
        return
    }

    response.Ok(c)
}

type AddJobRequest struct {
    Name string `json:"name" binding:"required"`
    Spec string `json:"spec" binding:"required"`
    Type string `json:"type" binding:"required"` // 任务类型，用于查找 fn
}
```

## pkg/xcron 要不要加 Manager？

### 现有极简版（推荐保持）

```go
// pkg/xcron/xcron.go
type Scheduler struct {
    *cron.Cron  // 嵌入，直接暴露原生 API
}

func New() *Scheduler {
    c := cron.New(
        cron.WithSeconds(),
        cron.WithLogger(slogAdapter{}),
        cron.WithChain(cron.Recover(slogAdapter{})),
    )
    return &Scheduler{Cron: c}
}

func (s *Scheduler) Start(ctx context.Context) error {
    s.Cron.Start()
    slog.Info("cron scheduler started", "jobs", len(s.Entries()))
    <-ctx.Done()
    slog.Info("cron scheduler stopping")
    <-s.Cron.Stop().Done()
    slog.Info("cron scheduler stopped")
    return nil
}
```

**优点**：
- 极简，职责单一（只是生命周期封装）
- 不维护额外状态，调用方直接用 robfig/cron API
- 跨项目复用零成本

**不足**：
- 需要调用方自己管理"名字 → EntryID"映射

### Manager 增强版（backend-rbac 的实现）

在 xcron 内部加 `map[string]*job` + 锁 + Add/Remove/List/GetSpec 等方法。

**优点**：
- 开箱即用，不需要调用方再写一层 service

**缺点**：
- xcron 职责膨胀（既管调度又管注册表）
- 多一层锁竞争（robfig/cron 内部有锁，Manager 又加一层）
- 错误消息、日志风格带项目特征（如中文），跨项目复用需改

**结论**：**本项目用极简版**，名字映射放 `internal/service/cronjob`（业务语义），保持 pkg/xcron 纯粹。

## 总结

| | 方案 A（字段） | 方案 B（平级） |
|---|---|---|
| 适合场景 | 单体应用，组件强绑定 | 微服务拆分预期强 |
| main.go 复杂度 | 低（Server 内部编排） | 中（main 编排多组件） |
| 独立部署 scheduler | 需重构 | 改 main 即可 |
| HTTP 访问 scheduler | 自然（同 struct） | 需传引用 |

**本项目推荐**：**方案 A**（scheduler 作为 Server 字段），理由是当前单体，且动态管理 API 需要访问 scheduler。
